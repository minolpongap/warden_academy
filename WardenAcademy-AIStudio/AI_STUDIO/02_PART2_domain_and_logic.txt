package pg.wardenacademy.domain

/**
 * Points rules (shown to trainees on the About screen):
 *  - 10 points per correct answer
 *  - Pass mark: 70%
 *  - +20 for passing on the FIRST attempt
 *  - +25 for a perfect score
 *  - Up to +15 speed bonus on a pass, in proportion to time left
 * A module counts toward the leaderboard by its BEST attempt only, so retakes
 * can improve a score but cannot be farmed for extra points.
 */
object Scoring {
    const val POINTS_PER_CORRECT = 10
    const val PASS_PERCENT = 70
    const val FIRST_ATTEMPT_PASS_BONUS = 20
    const val PERFECT_BONUS = 25
    const val MAX_SPEED_BONUS = 15
    const val SECONDS_PER_QUESTION = 45

    data class Result(
        val basePoints: Int,
        val bonusPoints: Int,
        val totalPoints: Int,
        val percent: Int,
        val passed: Boolean,
        val perfect: Boolean,
        val breakdown: List<Pair<String, Int>>
    )

    fun timeLimitFor(questionCount: Int): Int = questionCount * SECONDS_PER_QUESTION

    fun percent(correct: Int, total: Int): Int = if (total <= 0) 0 else correct * 100 / total

    fun maxPointsFor(questionCount: Int): Int =
        questionCount * POINTS_PER_CORRECT + FIRST_ATTEMPT_PASS_BONUS + PERFECT_BONUS + MAX_SPEED_BONUS

    fun score(
        correct: Int,
        total: Int,
        attemptNumber: Int,
        secondsUsed: Int,
        timeLimitSeconds: Int
    ): Result {
        val base = correct * POINTS_PER_CORRECT
        val pct = percent(correct, total)
        val passed = pct >= PASS_PERCENT
        val perfect = total > 0 && correct == total
        val breakdown = mutableListOf("Correct answers ($correct × $POINTS_PER_CORRECT)" to base)
        var bonus = 0

        if (passed && attemptNumber == 1) {
            bonus += FIRST_ATTEMPT_PASS_BONUS
            breakdown += "First-attempt pass bonus" to FIRST_ATTEMPT_PASS_BONUS
        }
        if (perfect) {
            bonus += PERFECT_BONUS
            breakdown += "Perfect score bonus" to PERFECT_BONUS
        }
        if (passed && timeLimitSeconds > 0) {
            val remaining = (timeLimitSeconds - secondsUsed).coerceIn(0, timeLimitSeconds)
            val speed = MAX_SPEED_BONUS * remaining / timeLimitSeconds
            if (speed > 0) {
                bonus += speed
                breakdown += "Speed bonus" to speed
            }
        }
        return Result(base, bonus, base + bonus, pct, passed, perfect, breakdown)
    }
}

package pg.wardenacademy.domain

import pg.wardenacademy.data.content.TrainingModule
import pg.wardenacademy.data.db.QuizAttempt
import pg.wardenacademy.data.db.Trainee

enum class Rank(val title: String, val minPoints: Int) {
    TRAINEE("Trainee", 0),
    FIELD_OFFICER("Field Officer", 100),
    ASSISTANT_WARDEN("Assistant Warden", 250),
    WARDEN("Warden", 450),
    SENIOR_WARDEN("Senior Warden", 650),
    CHIEF_WARDEN("Chief Warden", 850);

    val next: Rank? get() = Rank.entries.getOrNull(ordinal + 1)

    companion object {
        fun forPoints(points: Int): Rank = Rank.entries.last { points >= it.minPoints }
    }
}

enum class Badge(val title: String, val description: String, val emoji: String) {
    FIRST_STEPS("First Steps", "Pass your first module", "🥾"),
    HEARING_READY("Hearing Ready", "Pass Modules 1–4 (hearings and reports)", "🗣️"),
    COMPENSATION_EXPERT("Compensation Expert", "Pass Modules 5 and 6", "⚖️"),
    PERFECTIONIST("Perfectionist", "Score 100% on any quiz", "💯"),
    FLAWLESS_FIRST("Flawless First", "Score 100% on a first attempt", "🌟"),
    TRIPLE_PERFECT("Triple Perfect", "Score 100% on three different modules", "🏅"),
    FULLY_CERTIFIED("Fully Certified", "Pass every module", "🎓")
}

data class ModuleProgress(
    val module: TrainingModule,
    val attempts: Int,
    val bestPoints: Int,
    val bestPercent: Int,
    val passed: Boolean,
    val perfect: Boolean,
    val unlocked: Boolean
)

data class TraineeStats(
    val trainee: Trainee,
    val totalPoints: Int,
    val modulesPassed: Int,
    val totalModules: Int,
    val perfectModules: Int,
    val totalAttempts: Int,
    val averageBestPercent: Int,
    val rank: Rank,
    val badges: List<Badge>,
    val lastActivity: Long,
    val position: Int = 0
) {
    val allPassed: Boolean get() = totalModules > 0 && modulesPassed == totalModules
}

object Progress {

    /** Each module is unlocked once the module before it has been passed. */
    fun moduleProgress(modules: List<TrainingModule>, attempts: List<QuizAttempt>): List<ModuleProgress> {
        val byModule = attempts.groupBy { it.moduleId }
        var previousPassed = true
        return modules.sortedBy { it.number }.map { m ->
            val list = byModule[m.id].orEmpty()
            val best = list.maxWithOrNull(compareBy<QuizAttempt> { it.totalPoints }.thenBy { it.correct })
            val passed = list.any { it.passed }
            val mp = ModuleProgress(
                module = m,
                attempts = list.size,
                bestPoints = best?.totalPoints ?: 0,
                bestPercent = list.maxOfOrNull { Scoring.percent(it.correct, it.total) } ?: 0,
                passed = passed,
                perfect = list.any { it.perfect },
                unlocked = previousPassed
            )
            previousPassed = passed
            mp
        }
    }

    fun stats(trainee: Trainee, modules: List<TrainingModule>, attempts: List<QuizAttempt>): TraineeStats {
        val mine = attempts.filter { it.traineeId == trainee.id }
        val progress = moduleProgress(modules, mine)
        val total = progress.sumOf { it.bestPoints }
        val passedIds = progress.filter { it.passed }.map { it.module.number }.toSet()
        val perfectCount = progress.count { it.perfect }
        val attempted = progress.filter { it.attempts > 0 }

        val badges = buildList {
            if (passedIds.isNotEmpty()) add(Badge.FIRST_STEPS)
            if (passedIds.containsAll(listOf(1, 2, 3, 4))) add(Badge.HEARING_READY)
            if (passedIds.containsAll(listOf(5, 6))) add(Badge.COMPENSATION_EXPERT)
            if (perfectCount >= 1) add(Badge.PERFECTIONIST)
            if (mine.any { it.perfect && it.attemptNumber == 1 }) add(Badge.FLAWLESS_FIRST)
            if (perfectCount >= 3) add(Badge.TRIPLE_PERFECT)
            if (modules.isNotEmpty() && passedIds.size == modules.size) add(Badge.FULLY_CERTIFIED)
        }

        return TraineeStats(
            trainee = trainee,
            totalPoints = total,
            modulesPassed = passedIds.size,
            totalModules = modules.size,
            perfectModules = perfectCount,
            totalAttempts = mine.size,
            averageBestPercent = if (attempted.isEmpty()) 0 else attempted.sumOf { it.bestPercent } / attempted.size,
            rank = Rank.forPoints(total),
            badges = badges,
            lastActivity = mine.maxOfOrNull { it.completedAt } ?: trainee.createdAt
        )
    }

    /** Sort: points ↓, modules passed ↓, then whoever got there first. Ties share a position. */
    fun leaderboard(
        trainees: List<Trainee>,
        modules: List<TrainingModule>,
        attempts: List<QuizAttempt>
    ): List<TraineeStats> {
        val sorted = trainees.map { stats(it, modules, attempts) }
            .sortedWith(
                compareByDescending<TraineeStats> { it.totalPoints }
                    .thenByDescending { it.modulesPassed }
                    .thenBy { it.lastActivity }
            )
        var lastKey: Pair<Int, Int>? = null
        var lastPos = 0
        return sorted.mapIndexed { i, s ->
            val key = s.totalPoints to s.modulesPassed
            if (key != lastKey) { lastPos = i + 1; lastKey = key }
            s.copy(position = lastPos)
        }
    }
}

sealed interface Screen {
    data object Welcome : Screen
    data object Home : Screen
    data object Leaderboard : Screen
    data object Glossary : Screen
    data object Profile : Screen
    data class Module(val id: String) : Screen
    /** `session` makes every quiz visit a fresh attempt. */
    data class Quiz(val id: String, val session: Long = System.nanoTime()) : Screen
    data object Certificate : Screen
    data object About : Screen
}

val TAB_SCREENS: List<Screen> = listOf(Screen.Home, Screen.Leaderboard, Screen.Glossary, Screen.Profile)

class AppViewModel(private val app: WardenApp) : ViewModel() {

    val course: Course = app.content.course
    private val store = app.store

    // ---------- Navigation ----------
    var backStack by mutableStateOf(
        listOf(if (app.session.currentTraineeId == null) Screen.Welcome else Screen.Home)
    )
        private set
    val screen: Screen get() = backStack.last()
    val canGoBack: Boolean get() = backStack.size > 1

    fun navigate(to: Screen) { backStack = backStack + to }
    fun back() { if (canGoBack) backStack = backStack.dropLast(1) }
    fun openTab(tab: Screen) { backStack = if (tab == Screen.Home) listOf(Screen.Home) else listOf(Screen.Home, tab) }
    fun resetTo(to: Screen) { backStack = listOf(to) }

    // ---------- Data ----------
    private val _currentId = MutableStateFlow(app.session.currentTraineeId)
    val currentId: StateFlow<Long?> = _currentId

    val trainees: StateFlow<List<Trainee>> = store.trainees
    private val attempts: StateFlow<List<QuizAttempt>> = store.attempts

    val leaderboard: StateFlow<List<TraineeStats>> =
        combine(trainees, attempts) { t, a -> Progress.leaderboard(t, course.modules, a) }
            .stateIn(viewModelScope, SharingStarted.Eagerly, Progress.leaderboard(trainees.value, course.modules, attempts.value))

    val currentStats: StateFlow<TraineeStats?> =
        combine(_currentId, leaderboard) { id, board -> board.firstOrNull { it.trainee.id == id } }
            .stateIn(viewModelScope, SharingStarted.Eagerly, null)

    val myAttempts: StateFlow<List<QuizAttempt>> =
        combine(_currentId, attempts) { id, a -> a.filter { it.traineeId == id } }
            .stateIn(viewModelScope, SharingStarted.Eagerly, emptyList())

    val moduleProgress: StateFlow<List<ModuleProgress>> =
        myAttempts.map { Progress.moduleProgress(course.modules, it) }
            .stateIn(viewModelScope, SharingStarted.Eagerly, Progress.moduleProgress(course.modules, emptyList()))

    fun register(name: String, staffId: String, province: String, onResult: (String?) -> Unit) {
        val cleanName = name.trim()
        val cleanId = staffId.trim().uppercase()
        if (cleanName.length < 2) return onResult("Please enter your full name.")
        if (cleanId.isEmpty()) return onResult("Please enter your staff or trainee ID.")
        if (province.isBlank()) return onResult("Please choose your province.")
        val id = store.insertTrainee(Trainee(name = cleanName, staffId = cleanId, province = province))
            ?: return onResult("ID $cleanId is already registered on this device. Select it from the list.")
        select(id)
        onResult(null)
    }

    fun select(id: Long) {
        app.session.currentTraineeId = id
        _currentId.value = id
    }

    fun signOut() {
        app.session.currentTraineeId = null
        _currentId.value = null
        resetTo(Screen.Welcome)
    }

    fun deleteTrainee(trainee: Trainee) {
        store.deleteTrainee(trainee)
        if (_currentId.value == trainee.id) {
            app.session.currentTraineeId = null
            _currentId.value = null
        }
    }
}

/** A question with its options shuffled for this attempt. */
data class QuizItem(
    val question: Question,
    val options: List<String>,
    val correctIndex: Int
)

data class QuizUiState(
    val items: List<QuizItem> = emptyList(),
    val index: Int = 0,
    val selected: Map<Int, Int> = emptyMap(),   // question index -> chosen option index
    val checked: Set<Int> = emptySet(),          // questions whose answer has been revealed
    val timeLimit: Int = 0,
    val secondsLeft: Int = 0,
    val result: Scoring.Result? = null,
    val attemptNumber: Int = 0,
    val timedOut: Boolean = false
) {
    val current: QuizItem? get() = items.getOrNull(index)
    val correctCount: Int get() = items.indices.count { selected[it] == items[it].correctIndex }
    val finished: Boolean get() = result != null
}

/**
 * Runs one quiz. It is tied to the quiz screen's composition scope, so leaving the
 * screen cancels the timer and nothing is recorded.
 */
class QuizController(
    private val store: LocalStore,
    private val scope: CoroutineScope,
    private val traineeId: Long,
    val module: TrainingModule
) {
    private val _state = MutableStateFlow(QuizUiState())
    val state: StateFlow<QuizUiState> = _state
    private var timerJob: Job? = null

    init { start() }

    fun start() {
        val items = module.quiz.shuffled().map { q ->
            val order = q.options.indices.shuffled()
            QuizItem(q, order.map { q.options[it] }, order.indexOf(q.answerIndex))
        }
        val limit = Scoring.timeLimitFor(items.size)
        _state.value = QuizUiState(items = items, timeLimit = limit, secondsLeft = limit)
        timerJob?.cancel()
        timerJob = scope.launch {
            while (isActive && _state.value.secondsLeft > 0 && !_state.value.finished) {
                delay(1_000)
                _state.update { it.copy(secondsLeft = (it.secondsLeft - 1).coerceAtLeast(0)) }
            }
            if (!_state.value.finished) {
                _state.update { it.copy(timedOut = true) }
                finish()
            }
        }
    }

    fun select(option: Int) {
        val s = _state.value
        if (s.finished || s.index in s.checked) return
        _state.update { it.copy(selected = it.selected + (it.index to option)) }
    }

    fun check() {
        if (_state.value.index !in _state.value.selected) return
        _state.update { it.copy(checked = it.checked + it.index) }
    }

    fun next() {
        val s = _state.value
        if (s.index < s.items.lastIndex) _state.update { it.copy(index = it.index + 1) } else finish()
    }

    fun finish() {
        val s = _state.value
        if (s.finished) return
        timerJob?.cancel()
        val attemptNumber = store.countAttempts(traineeId, module.id) + 1
        val used = s.timeLimit - s.secondsLeft
        val result = Scoring.score(s.correctCount, s.items.size, attemptNumber, used, s.timeLimit)
        store.insertAttempt(
            QuizAttempt(
                traineeId = traineeId, moduleId = module.id, attemptNumber = attemptNumber,
                correct = s.correctCount, total = s.items.size, secondsUsed = used,
                timeLimitSeconds = s.timeLimit, basePoints = result.basePoints,
                bonusPoints = result.bonusPoints, totalPoints = result.totalPoints,
                passed = result.passed, perfect = result.perfect
            )
        )
        _state.update { it.copy(result = result, attemptNumber = attemptNumber) }
    }
}

    Tab(Screen.Leaderboard, "Leaderboard", Icons.Filled.EmojiEvents),
    Tab(Screen.Glossary, "Glossary", Icons.Filled.MenuBook),
    Tab(Screen.Profile, "Profile", Icons.Filled.AccountCircle),

/** Single-activity root: bottom tabs plus a ViewModel-held back stack (no Navigation library). */
@Composable
fun WardenRoot(app: WardenApp) {
    val vm: AppViewModel = viewModel(factory = viewModelFactory { initializer { AppViewModel(app) } })
    val screen = vm.screen
    val showTabs = screen in TAB_SCREENS

    // The quiz screen handles its own back press (it asks before leaving).
    BackHandler(enabled = vm.canGoBack && screen !is Screen.Quiz) { vm.back() }

    Scaffold(
        contentWindowInsets = WindowInsets(0),
        bottomBar = {
            if (showTabs) {
                NavigationBar {
                    tabs.forEach { tab ->
                        NavigationBarItem(
                            selected = screen == tab.screen,
                            onClick = { vm.openTab(tab.screen) },
                            icon = { Icon(tab.icon, contentDescription = tab.label) },
                            label = { Text(tab.label) }
                        )
                    }
                }
            }
        }
    ) { padding ->
        Box(Modifier.padding(bottom = padding.calculateBottomPadding())) {
            when (screen) {
                Screen.Welcome -> WelcomeScreen(vm) { vm.resetTo(Screen.Home) }
                Screen.Home -> HomeScreen(vm, onOpenModule = { vm.navigate(Screen.Module(it)) })
                Screen.Leaderboard -> LeaderboardScreen(vm)
                Screen.Glossary -> GlossaryScreen(vm)
                Screen.Profile -> ProfileScreen(
                    vm = vm,
                    onCertificate = { vm.navigate(Screen.Certificate) },
                    onAbout = { vm.navigate(Screen.About) },
                    onSwitchTrainee = { vm.signOut() }
                )
                is Screen.Module -> ModuleScreen(
                    vm = vm,
                    moduleId = screen.id,
                    onBack = { vm.back() },
                    onStartQuiz = { vm.navigate(Screen.Quiz(screen.id)) }
                )
                is Screen.Quiz -> QuizScreen(
                    app = app,
                    vm = vm,
                    quiz = screen,
                    onExit = { vm.back() },
                    onLeaderboard = { vm.openTab(Screen.Leaderboard) }
                )
                Screen.Certificate -> CertificateScreen(vm, onBack = { vm.back() })
                Screen.About -> AboutScreen(vm, onBack = { vm.back() })
            }
        }
    }
}

package pg.wardenacademy.util

import android.content.Context
import android.content.Intent
import android.graphics.Color
import android.graphics.Paint
import android.graphics.RectF
import android.graphics.Typeface
import android.graphics.pdf.PdfDocument
import androidx.core.content.FileProvider
import java.io.File
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

/** Draws an A4-landscape certificate with the platform PdfDocument API (no extra libraries). */
object CertificatePdf {

    data class Details(
        val name: String,
        val staffId: String,
        val province: String,
        val rank: String,
        val points: Int,
        val modules: Int,
        val averagePercent: Int,
        val issuedAt: Long = System.currentTimeMillis()
    )

    fun create(context: Context, d: Details): File {
        val width = 842   // A4 landscape, in points
        val height = 595
        val doc = PdfDocument()
        val page = doc.startPage(PdfDocument.PageInfo.Builder(width, height, 1).create())
        val c = page.canvas

        val red = Color.rgb(183, 28, 28)
        val gold = Color.rgb(249, 168, 37)
        val black = Color.rgb(26, 26, 26)

        // Borders
        val border = Paint(Paint.ANTI_ALIAS_FLAG).apply { style = Paint.Style.STROKE }
        border.color = red; border.strokeWidth = 14f
        c.drawRect(RectF(14f, 14f, width - 14f, height - 14f), border)
        border.color = gold; border.strokeWidth = 4f
        c.drawRect(RectF(32f, 32f, width - 32f, height - 32f), border)

        fun text(s: String, y: Float, size: Float, color: Int = black, bold: Boolean = false, italic: Boolean = false) {
            val p = Paint(Paint.ANTI_ALIAS_FLAG).apply {
                this.color = color
                textSize = size
                textAlign = Paint.Align.CENTER
                typeface = Typeface.create(
                    Typeface.SERIF,
                    when {
                        bold && italic -> Typeface.BOLD_ITALIC
                        bold -> Typeface.BOLD
                        italic -> Typeface.ITALIC
                        else -> Typeface.NORMAL
                    }
                )
            }
            c.drawText(s, width / 2f, y, p)
        }

        text("WARDEN ACADEMY PNG", 92f, 18f, red, bold = true)
        text("Certificate of Completion", 140f, 40f, black, bold = true)
        text("This is to certify that", 190f, 16f, italic = true)
        text(d.name, 240f, 36f, red, bold = true)
        text("Staff ID ${d.staffId}  •  ${d.province}", 268f, 13f)
        text("has completed all ${d.modules} modules of the training programme", 310f, 16f)
        text("Functions of the Mining Warden under the Mining Act 1992 (Papua New Guinea)", 336f, 16f, bold = true)
        text("Rank achieved: ${d.rank}   •   Points: ${d.points}   •   Average best score: ${d.averagePercent}%", 378f, 14f)

        val date = SimpleDateFormat("d MMMM yyyy", Locale.getDefault()).format(Date(d.issuedAt))
        text("Issued $date", 412f, 13f, italic = true)

        // Signature lines
        val line = Paint().apply { color = black; strokeWidth = 1f }
        c.drawLine(140f, 490f, 340f, 490f, line)
        c.drawLine(width - 340f, 490f, width - 140f, 490f, line)
        val small = Paint(Paint.ANTI_ALIAS_FLAG).apply { color = black; textSize = 11f; textAlign = Paint.Align.CENTER }
        c.drawText("Training Coordinator", 240f, 506f, small)
        c.drawText("Chief Warden", width - 240f, 506f, small)

        val ref = "Certificate ref: WA-${d.staffId}-${d.issuedAt / 1000}"
        c.drawText(ref, width / 2f, 548f, small.apply { color = Color.GRAY; textSize = 9f })

        doc.finishPage(page)
        val dir = File(context.cacheDir, "certificates").apply { mkdirs() }
        val safeName = d.name.replace(Regex("[^A-Za-z0-9]+"), "_")
        val file = File(dir, "Warden_Certificate_$safeName.pdf")
        file.outputStream().use { doc.writeTo(it) }
        doc.close()
        return file
    }

    fun share(context: Context, file: File) {
        val uri = FileProvider.getUriForFile(context, "${context.packageName}.fileprovider", file)
        val intent = Intent(Intent.ACTION_SEND).apply {
            type = "application/pdf"
            putExtra(Intent.EXTRA_STREAM, uri)
            addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
        }
        context.startActivity(Intent.createChooser(intent, "Share certificate"))
    }
}
